                    let finalString = item;

                    

                    if (qty) finalString = `${qty}x ${finalString}`;

                    if (sn) finalString = `${finalString} (SN: ${sn})`;

                    

                    selectedItems.push(finalString);

                }

            });


            if (selectedItems.length === 0) {

                alert("System Error: Container is empty. Please select at least one item.");

                return;

            }


            containersData.push({

                title: `${boxType} ${boxNumber}` + (boxOwner ? ` - ${boxOwner}` : ''),

                items: selectedItems

            });


            document.getElementById('boxNumber').value = parseInt(boxNumber) + 1;

            masterEquipment.forEach((_, index) => {

                const checkEl = document.getElementById(`item-${index}_check`);

                if(checkEl) {

                    checkEl.checked = false;

                    document.getElementById(`item-${index}_qty`).value = '';

                    document.getElementById(`item-${index}_sn`).value = '';

                }

            });


            updatePreview();

        }


        function updatePreview() {

            const previewArea = document.getElementById('previewArea');

            const pdfContainers = document.getElementById('pdf-containers');

            

            if (containersData.length > 0 && previewArea.innerHTML.includes('NO CONTAINERS LOADED')) {

                previewArea.innerHTML = ''; 

            }

            

            previewArea.innerHTML = '';

            pdfContainers.innerHTML = '';


            containersData.forEach(container => {

                const containerHTML = `

                    <div class="mb-5 bg-slate-800/80 p-3 rounded-lg border border-slate-700 shadow">

                        <h2 class="font-bold text-amber-500 text-md mb-2 flex items-center">

                            <span class="bg-amber-500 text-slate-900 px-2 py-0.5 rounded-sm mr-2 text-xs">BOX</span>

                            ${container.title}

                        </h2>

                        <ul class="list-none space-y-1 text-slate-300 text-sm">

                            ${container.items.map(item => `

                                <li class="flex items-start">

                                    <span class="text-amber-500 mr-2 mt-0.5">▸</span>

                                    <span>${item}</span>

                                </li>`).join('')}

                        </ul>

                    </div>

                `;

                previewArea.innerHTML += containerHTML;

                

                const pdfSection = document.createElement('div');

                pdfSection.innerHTML = `<h2>${container.title}</h2><ul>${container.items.map(item => `<li>${item}</li>`).join('')}</ul>`;

                pdfContainers.appendChild(pdfSection);

            });

        }
